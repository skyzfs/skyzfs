// Next look at #157 (head 4edc1b3): a scratch test appended to
// crates/skyzfs-agent/src/engine/tests.rs for one run and then removed.
// Not product code. Through the PR's ThroughCli wrapper (the real
// ZfsCli adapter), a property that a person sets as each take lands is
// still compared, whatever its name: a user property whose name ends in
// the counters' names, and the limits themselves. Each take then fails
// its check and the publish is given up NotWhole at the third, as
// `a_take_not_as_before_names_the_datasets_that_differ` shows for a
// property read without the adapter. The control row (`None`) changes
// nothing and is published.

#[test]
fn review_nl157_a_property_a_person_sets_is_still_compared() {
    for property in [
        None,
        Some("com.example:snapshot_count"),
        Some("x:filesystem_count"),
        Some("snapshot_limit"),
        Some("filesystem_limit"),
    ] {
        let world = World::new(Mode::Busy);
        let keep = MemKeep::default();
        let n = Rc::new(Cell::new(0u32));
        let points: Points = Box::new({
            let (world, n) = (world.clone(), n.clone());
            move |p: Point, _: &str| {
                // Between the read before the take and the take.
                if p == at(Effect::Take, false) {
                    n.set(n.get() + 1);
                    if let Some(property) = property {
                        let mut w = world.borrow_mut();
                        if let Some(pool) = &mut w.pool {
                            if let Some(db) =
                                pool.datasets.iter_mut().find(|d| d.name.ends_with("/db"))
                            {
                                db.local.insert(
                                    property.to_string(),
                                    (100 + n.get()).to_string().into_bytes(),
                                );
                            }
                        }
                    }
                }
                false
            }
        });
        let mut agent = Agent::new(
            ThroughCli(FakeZfs(world.clone())),
            FakeDaemon(world.clone()),
            keep.clone(),
            FakeClock(world.clone()),
            VOLUME,
            POOL,
        )
        .expect("the agent")
        .with_points(points);
        agent.begin().expect("begun");
        let settled = agent.settle(LIMIT);
        let outcomes = agent.outcomes().to_vec();
        match property {
            None => assert!(
                matches!(outcomes[..], [(_, Outcome::Published(_))]) && n.get() == 1,
                "control: ok={} {outcomes:?} takes {}",
                settled.is_ok(),
                n.get()
            ),
            Some(property) => assert!(
                matches!(outcomes[..], [(_, Outcome::Abandoned(Why::NotWhole))])
                    && n.get() == plan::CHECKS_FAILED_MAX,
                "{property}: ok={} {outcomes:?} takes {}",
                settled.is_ok(),
                n.get()
            ),
        }
    }
}
