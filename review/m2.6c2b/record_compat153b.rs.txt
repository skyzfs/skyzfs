/// Review scratch: a record written by 9717321 (version 2 without
/// `Why::NotWhole`) reads at this head as the same record, and the
/// attempts of a publish at Intent written then (always 0 there) read as
/// no failed check.
#[test]
fn review_scratch_the_previous_heads_record_reads_the_same() {
    let hex: String = OLD_GOLDEN.concat();
    let bytes: Vec<u8> = (0..hex.len())
        .step_by(2)
        .map(|at| u8::from_str_radix(&hex[at..at + 2], 16).expect("hex"))
        .collect();
    let mut want = golden();
    want.abandoned.retain(|a| a.why != Why::NotWhole);
    assert_eq!(Record::decode(&bytes), Ok(want.clone()));
    assert_eq!(want.encode().expect("encoded"), bytes);
    println!("REVIEW old golden decodes at this head: ok; {} bytes", bytes.len());
}
